   338→function randomGameId(): string {
   339→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) {
   340→    return `game_${crypto.randomUUID().slice(0, 8)}`
   341→  }
   342→  return `game_${Math.random().toString(36).slice(2, 10)}`
   343→}

   345→function statusColor(status: string): string {
   346→  switch (status) {
   347→    case 'healthy':
   348→      return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   349→    case 'degraded':
   350→      return 'text-amber-400 bg-amber-500/10 border-amber-500/30'
   351→    case 'unhealthy':
   352→      return 'text-rose-400 bg-rose-500/10 border-rose-500/30'
   353→    default:
   354→      return 'text-zinc-400 bg-zinc-500/10 border-zinc-500/30'
   355→  }
   356→}

   805→type StepStatus = 'pending' | 'running' | 'success' | 'error'

   807→interface StepState {
   808→  status: StepStatus
   809→  result?: unknown
   810→  error?: string
   811→  timestamp?: string
   812→}

   814→function EventSourcingDemo() {
   815→  const { toast } = useToast()
   816→  const [gameId, setGameId] = React.useState('')
   817→  const [title, setTitle] = React.useState('')
   818→  const [creatorId, setCreatorId] = React.useState('')
   819→  const [steps, setSteps] = React.useState<Record<number, StepState>>({
   820→    1: { status: 'pending' },
   821→    2: { status: 'pending' },
   822→    3: { status: 'pending' },
   823→    4: { status: 'pending' },
   824→  })
   825→
   826→  React.useEffect(() => {
   827→    setGameId(randomGameId())
   828→    setCreatorId(`creator_${Math.random().toString(36).slice(2, 8)}`)
   829→    setTitle('Liquid Tournament — Season 1')
   830→  }, [])
   831→
   832→  const updateStep = (n: number, patch: Partial<StepState>) =>
   833→    setSteps((s) => ({ ...s, [n]: { ...s[n], ...patch } }))
   834→
   835→  const resetSteps = (keepCommand = false) =>
   836→    setSteps({
   837→      1: keepCommand ? steps[1] : { status: 'pending' },
   838→      2: { status: 'pending' },
   839→      3: { status: 'pending' },
   840→      4: { status: 'pending' },
   841→    })
   842→
   843→  // Step 1: Dispatch command
   844→  const handleDispatch = async () => {
   845→    if (!gameId.trim() || !title.trim() || !creatorId.trim()) {
   846→      toast({
   847→        title: 'Validation error',
   848→        description: 'Game ID, Title, and Creator ID are all required.',
   849→        variant: 'destructive',
   850→      })
   851→      return
   852→    }
   853→    updateStep(1, { status: 'running', error: undefined, result: undefined })
   854→    try {
   855→      const res = await fetch('/api/commands', {
   856→        method: 'POST',
   857→        headers: { 'Content-Type': 'application/json' },
   858→        body: JSON.stringify({
   859→          commandType: 'PublishGame',
   860→          payload: { gameId, title, creatorId },
   861→        }),
   862→      })
   863→      const data = (await res.json()) as CommandResponse
   864→      if (!data.ok) {
   865→        throw new Error(data.error ?? 'Command failed')
   866→      }
   867→      updateStep(1, {
   868→        status: 'success',
   869→        result: data.data,
   870→        timestamp: new Date().toISOString(),
   871→      })
   872→      // Reset downstream steps since we have new events
   873→      setSteps((s) => ({
   874→        1: s[1],
   875→        2: { status: 'pending' },
   876→        3: { status: 'pending' },
   877→        4: { status: 'pending' },
   878→      }))
   879→      toast({
   880→        title: 'Command dispatched',
   881→        description: `PublishGame appended to stream GameAggregate:${gameId}`,
   882→      })
   883→    } catch (e) {
   884→      const msg = e instanceof Error ? e.message : 'Unknown error'
   885→      updateStep(1, { status: 'error', error: msg })
   886→      toast({ title: 'Command failed', description: msg, variant: 'destructive' })
   887→    }
   888→  }
   889→
   890→  // Step 2: Process outbox
   891→  const handleOutbox = async () => {
   892→    updateStep(2, { status: 'running', error: undefined, result: undefined })
   893→    try {
   894→      const res = await fetch('/api/workers/outbox', { method: 'POST' })
   895→      const data = (await res.json()) as OutboxResponse
   896→      updateStep(2, {
   897→        status: 'success',
   898→        result: data,
   899→        timestamp: new Date().toISOString(),
   900→      })
   901→      toast({
   902→        title: 'Outbox processed',
   903→        description: `Published ${data.published} event(s).`,
   904→      })
   905→    } catch (e) {
   906→      const msg = e instanceof Error ? e.message : 'Unknown error'
   907→      updateStep(2, { status: 'error', error: msg })
   908→      toast({ title: 'Outbox failed', description: msg, variant: 'destructive' })
   909→    }
   910→  }
   911→
   912→  // Step 3: Run projections
   913→  const handleProjections = async () => {
   914→    updateStep(3, { status: 'running', error: undefined, result: undefined })
   915→    try {
   916→      const res = await fetch('/api/workers/projections', { method: 'POST' })
   917→      const data = (await res.json()) as ProjectionsResponse
   918→      updateStep(3, {
   919→        status: 'success',
   920→        result: data,
   921→        timestamp: new Date().toISOString(),
   922→      })
   923→      toast({
   924→        title: 'Projections ran',
   925→        description: `Projected ${data.processed} event(s) into read models.`,
   926→      })
   927→    } catch (e) {
   928→      const msg = e instanceof Error ? e.message : 'Unknown error'
   929→      updateStep(3, { status: 'error', error: msg })
   930→      toast({
   931→        title: 'Projections failed',
   932→        description: msg,
   933→        variant: 'destructive',
   934→      })
   935→    }
   936→  }
   937→
   938→  // Step 4: Query read model
   939→  const handleQuery = async () => {
   940→    updateStep(4, { status: 'running', error: undefined, result: undefined })
   941→    try {
   942→      const res = await fetch('/api/queries', {
   943→        method: 'POST',
   944→        headers: { 'Content-Type': 'application/json' },
   945→        body: JSON.stringify({
   946→          queryType: 'GetGame',
   947→          payload: { gameId },
   948→        }),
   949→      })
   950→      const data = (await res.json()) as QueryResponse
   951→      if (!data.ok) {
   952→        throw new Error(data.error ?? 'Query failed')
   953→      }
   954→      updateStep(4, {
   955→        status: 'success',
   956→        result: data.data,
   957→        timestamp: new Date().toISOString(),
   958→      })
   959→      toast({
   960→        title: 'Query executed',
   961→        description: `Read model returned for game ${gameId}.`,
   962→      })
   963→    } catch (e) {
   964→      const msg = e instanceof Error ? e.message : 'Unknown error'
   965→      updateStep(4, { status: 'error', error: msg })
   966→      toast({ title: 'Query failed', description: msg, variant: 'destructive' })
   967→    }
   968→  }

   972→    setCreatorId(`creator_${Math.random().toString(36).slice(2, 8)}`)

   981→        title="Event Sourcing Demo"

   983→        icon={Zap}

   985→          <Button

   995→      />
   996→
   997→      {/* Flow indicator */}
   998→      <FlowIndicator steps={steps} />
   999→
  1000→      <div className="mt-6 grid gap-4 lg:grid-cols-2">
