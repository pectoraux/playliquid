   169→function randomGameId(): string {
   170→  if (typeof crypto !== 'undefined' && 'randomUUID' in crypto) {
   171→    return `game_${crypto.randomUUID().slice(0, 8)}`
   172→  }
   173→  return `game_${Math.random().toString(36).slice(2, 10)}`

   176→function statusColor(status: string): string {
   177→  switch (status) {
   178→    case 'healthy':
   179→      return 'text-emerald-400 bg-emerald-500/10 border-emerald-500/30'
   180→    case 'degraded':
   181→      return 'text-amber-400 bg-amber-500/10 border-amber-500/30'

   409→function HealthDashboard() {
   410→  const { toast } = useToast()
   411→  const [health, setHealth] = React.useState<HealthResponse | null>(null)
   412→  const [error, setError] = React.useState<string | null>(null)
   413→  const [loading, setLoading] = React.useState(true)
   414→  const [lastUpdated, setLastUpdated] = React.useState<Date | null>(null)
   415→  const [paused, setPaused] = React.useState(false)
   416→
   417→  const fetchHealth = React.useCallback(async () => {
   418→    try {
   419→      const res = await fetch('/api/health', { cache: 'no-store' })
   420→      if (!res.ok) {
   421→        throw new Error(`HTTP ${res.status}`)
   422→      }
   423→      const data = (await res.json()) as HealthResponse
   424→      setHealth(data)
   425→      setError(null)
   426→      setLastUpdated(new Date())
   427→    } catch (e) {
   428→      const msg = e instanceof Error ? e.message : 'Failed to fetch health'
   429→      setError(msg)
   430→    } finally {
   431→      setLoading(false)
   432→    }
   433→  }, [])
   434→
   435→  React.useEffect(() => {
   436→    void fetchHealth()
   437→    if (paused) return
   438→    const interval = setInterval(() => void fetchHealth(), 5000)
   439→    return () => clearInterval(interval)
   440→  }, [fetchHealth, paused])
   441→
   442→  const overall = health?.status ?? 'unhealthy'
   443→
   444→  const handleRefresh = () => {
   445→    setLoading(true)
   446→    void fetchHealth()
   447→    toast({
   448→      title: 'Health refreshed',
   449→      description: 'Re-ran all infrastructure health checks.',
   450→    })
   451→  }
   452→
   453→  return (
   454→    <section id="health" className="scroll-mt-8">
   455→      <SectionHeading
   456→        index="02"
   457→        title="Live Health Dashboard"
   458→        description="Real-time status of every infrastructure component. Auto-refreshes every 5 seconds."
   459→        icon={Activity}
   460→        action={
   461→          <div className="flex items-center gap-2">
   462→            <Button
   463→              variant="outline"
   464→              size="sm"
   465→              onClick={() => setPaused((p) => !p)}
   466→              className="border-slate-700 bg-slate-900/60 font-mono text-xs text-slate-300 hover:bg-slate-800 hover:text-zinc-100"
   467→            >
   468→              {paused ? 'Resume' : 'Pause'}
   469→            </Button>
   470→            <Button
   471→              variant="outline"
   472→              size="sm"
   473→              onClick={handleRefresh}
   474→              disabled={loading}
   475→              className="border-slate-700 bg-slate-900/60 font-mono text-xs text-slate-300 hover:bg-slate-800 hover:text-zinc-100"
   476→            >
   477→              <RefreshCw className={`h-3.5 w-3.5 ${loading ? 'animate-spin' : ''}`} />
   478→              Refresh
   479→            </Button>
   480→          </div>
   481→        }
   482→      />
   483→
   484→      <div className="mt-8 grid gap-4 lg:grid-cols-[1.2fr_2fr]">
   485→        {/* Overall status */}
   486→        <Card className="border-slate-800 bg-slate-900/50 shadow-lg shadow-black/20 ring-1 ring-inset ring-slate-800/60">
   487→          <CardHeader>
   488→            <div className="flex items-center justify-between">
   489→              <CardTitle className="font-mono text-sm uppercase tracking-wider text-slate-300">
   490→                Overall Status
   491→              </CardTitle>
   492→              <StatusDot status={overall} />
   493→            </div>
   494→            <CardDescription className="text-xs">
   495→              Aggregated from all registered health checks.
   496→            </CardDescription>
   497→          </CardHeader>
   498→          <CardContent className="space-y-4">
   499→            <div
   500→              className={`rounded-lg border px-4 py-3 ${statusColor(overall)}`}
   501→            >
   502→              <div className="flex items-center justify-between">
   503→                <span className="font-mono text-2xl font-bold uppercase">
   504→                  {loading && !health ? '...' : overall}
   505→                </span>
   506→                <Activity className="h-5 w-5 opacity-70" />
   507→              </div>
   508→            </div>
   509→            <div className="grid grid-cols-2 gap-2 font-mono text-xs">
   510→              <div className="rounded-md border border-slate-800 bg-slate-950/60 px-3 py-2">
   511→                <div className="text-slate-500">Checks</div>
   512→                <div className="text-zinc-200">{health?.checks.length ?? 0}</div>
   513→              </div>
   514→              <div className="rounded-md border border-slate-800 bg-slate-950/60 px-3 py-2">
   515→                <div className="text-slate-500">Last Updated</div>
   516→                <div className="text-zinc-200">
   517→                  {lastUpdated ? lastUpdated.toLocaleTimeString() : '—'}
   518→                </div>
   519→              </div>
   520→            </div>
   521→            {error && (
   522→              <div className="flex items-start gap-2 rounded-md border border-rose-500/30 bg-rose-500/10 px-3 py-2 text-xs text-rose-300">
   523→                <AlertCircle className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   524→                <span className="font-mono">{error}</span>
   525→              </div>
   526→            )}
   527→          </CardContent>
   528→        </Card>
   529→
   530→        {/* Per-component */}
   531→        <div className="grid gap-4 sm:grid-cols-2 xl:grid-cols-3">
   532→          {loading && !health ? (
   533→            <HealthCardSkeletons />
   534→          ) : health && health.checks.length > 0 ? (
   535→            health.checks.map((check) => (
   536→              <HealthCheckCard key={check.name} check={check} />
   537→            ))
   538→          ) : (
   539→            <div className="col-span-full flex items-center justify-center rounded-lg border border-dashed border-slate-800 py-12 text-sm text-slate-500">
   540→              No health checks registered.
   541→            </div>
   542→          )}
   543→        </div>
   544→      </div>
   545→    </section>
   546→  )
   547→}

   635→  }, [])

   637→  const updateStep = (n: number, patch: Partial<StepState>) =>
   638→    setSteps((s) => ({ ...s, [n]: { ...s[n], ...patch } }))
   639→
   640→  const resetSteps = (keepCommand = false) =>
   641→    setSteps({
   642→      1: keepCommand ? steps[1] : { status: 'pending' },
   643→      2: { status: 'pending' },
   644→      3: { status: 'pending' },
   645→      4: { status: 'pending' },

   647→
   648→  // Step 1: Dispatch command
   649→  const handleDispatch = async () => {
   650→    if (!gameId.trim() || !title.trim() || !creatorId.trim()) {
   651→      toast({
   652→        title: 'Validation error',
   653→        description: 'Game ID, Title, and Creator ID are all required.',
   654→        variant: 'destructive',
   655→      })
   656→      return
   657→    }
   658→    updateStep(1, { status: 'running', error: undefined, result: undefined })
   659→    try {
   660→      const res = await fetch('/api/commands', {
   661→        method: 'POST',
   662→        headers: { 'Content-Type': 'application/json' },
   663→        body: JSON.stringify({
   664→          commandType: 'PublishGame',
   665→          payload: { gameId, title, creatorId },
   666→        }),
   667→      })
   668→      const data = (await res.json()) as CommandResponse
   669→      if (!data.ok) {
   670→        throw new Error(data.error ?? 'Command failed')
   671→      }
   672→      updateStep(1, {
   673→        status: 'success',
   674→        result: data.data,
   675→        timestamp: new Date().toISOString(),
   676→      })
   677→      // Reset downstream steps since we have new events
   678→      setSteps((s) => ({
   679→        1: s[1],

   689→      const msg = e instanceof Error ? e.message : 'Unknown error'

   691→      toast({ title: 'Command failed', description: msg, variant: 'destructive' })
   692→    }
   693→  }
