   409→   409→function HealthDashboard() {
   410→   410→  const { toast } = useToast()
   411→   411→  const [health, setHealth] = React.useState<HealthResponse | null>(null)
   412→   412→  const [error, setError] = React.useState<string | null>(null)
   413→   413→  const [loading, setLoading] = React.useState(true)
   414→   414→  const [lastUpdated, setLastUpdated] = React.useState<Date | null>(null)
   415→   415→  const [paused, setPaused] = React.useState(false)
   416→   416→
   417→   417→  const fetchHealth = React.useCallback(async () => {
   418→   418→    try {
   419→   419→      const res = await fetch('/api/health', { cache: 'no-store' })
   420→   420→      if (!res.ok) {
   421→   421→        throw new Error(`HTTP ${res.status}`)
   422→   422→      }
   423→   423→      const data = (await res.json()) as HealthResponse
   424→   424→      setHealth(data)
   425→   425→      setError(null)
   426→   426→      setLastUpdated(new Date())
   427→   427→    } catch (e) {
   428→   428→      const msg = e instanceof Error ? e.message : 'Failed to fetch health'
   429→   429→      setError(msg)
   430→   430→    } finally {
   431→   431→      setLoading(false)
   432→   432→    }
   433→   433→  }, [])
   434→   434→
   435→   435→  React.useEffect(() => {
   436→   436→    void fetchHealth()
   437→   437→    if (paused) return
   438→   438→    const interval = setInterval(() => void fetchHealth(), 5000)
   439→   439→    return () => clearInterval(interval)
   440→   440→  }, [fetchHealth, paused])
   441→   441→
   442→   442→  const overall = health?.status ?? 'unhealthy'
   443→   443→
   444→   444→  const handleRefresh = () => {
   445→   445→    setLoading(true)
   446→   446→    void fetchHealth()
   447→   447→    toast({
   448→   448→      title: 'Health refreshed',
   449→   449→      description: 'Re-ran all infrastructure health checks.',
   450→   450→    })
   451→   451→  }
   452→   452→
   453→   453→  return (
   454→   454→    <section id="health" className="scroll-mt-8">
   455→   455→      <SectionHeading
   456→   456→        index="02"
   457→   457→        title="Live Health Dashboard"
   458→   458→        description="Real-time status of every infrastructure component. Auto-refreshes every 5 seconds."
   459→   459→        icon={Activity}
   460→   460→        action={
   461→   461→          <div className="flex items-center gap-2">
   462→   462→            <Button
   463→   463→              variant="outline"
   464→   464→              size="sm"
   465→   465→              onClick={() => setPaused((p) => !p)}
   466→   466→              className="border-slate-700 bg-slate-900/60 font-mono text-xs text-slate-300 hover:bg-slate-800 hover:text-zinc-100"
   467→   467→            >
   468→   468→              {paused ? 'Resume' : 'Pause'}
   469→   469→            </Button>
   470→   470→            <Button
   471→   471→              variant="outline"
   472→   472→              size="sm"

   610→   610→type StepStatus = 'pending' | 'running' | 'success' | 'error'

   612→   612→interface StepState {

   619→   619→function EventSourcingDemo() {
   620→   620→  const { toast } = useToast()
   621→   621→  const [gameId, setGameId] = React.useState('')
   622→   622→  const [title, setTitle] = React.useState('')
   623→   623→  const [creatorId, setCreatorId] = React.useState('')
   624→   624→  const [steps, setSteps] = React.useState<Record<number, StepState>>({
   625→   625→    1: { status: 'pending' },
   626→   626→    2: { status: 'pending' },
   627→   627→    3: { status: 'pending' },
   628→   628→    4: { status: 'pending' },
   629→   629→  })
   630→   630→
   631→   631→  React.useEffect(() => {
   632→   632→    setGameId(randomGameId())
   633→   633→    setCreatorId(`creator_${Math.random().toString(36).slice(2, 8)}`)
   634→   634→    setTitle('Liquid Tournament — Season 1')
   635→   635→  }, [])
   636→   636→
   637→   637→  const updateStep = (n: number, patch: Partial<StepState>) =>
   638→   638→    setSteps((s) => ({ ...s, [n]: { ...s[n], ...patch } }))
   639→   639→
   640→   640→  const resetSteps = (keepCommand = false) =>
   641→   641→    setSteps({
   642→   642→      1: keepCommand ? steps[1] : { status: 'pending' },
   643→   643→      2: { status: 'pending' },
   644→   644→      3: { status: 'pending' },
   645→   645→      4: { status: 'pending' },
   646→   646→    })
   647→   647→
   648→   648→  // Step 1: Dispatch command
   649→   649→  const handleDispatch = async () => {
   650→   650→    if (!gameId.trim() || !title.trim() || !creatorId.trim()) {
   651→   651→      toast({
   652→   652→        title: 'Validation error',
   653→   653→        description: 'Game ID, Title, and Creator ID are all required.',
   654→   654→        variant: 'destructive',
   655→   655→      })
   656→   656→      return
   657→   657→    }
   658→   658→    updateStep(1, { status: 'running', error: undefined, result: undefined })
   659→   659→    try {
   660→   660→      const res = await fetch('/api/commands', {
   661→   661→        method: 'POST',
   662→   662→        headers: { 'Content-Type': 'application/json' },
   663→   663→        body: JSON.stringify({
   664→   664→          commandType: 'PublishGame',
   665→   665→          payload: { gameId, title, creatorId },
   666→   666→        }),
   667→   667→      })
   668→   668→      const data = (await res.json()) as CommandResponse
   669→   669→      if (!data.ok) {
   670→   670→        throw new Error(data.error ?? 'Command failed')
   671→   671→      }
   672→   672→      updateStep(1, {
   673→   673→        status: 'success',
   674→   674→        result: data.data,
   675→   675→        timestamp: new Date().toISOString(),
   676→   676→      })
   677→   677→      // Reset downstream steps since we have new events
   678→   678→      setSteps((s) => ({
   679→   679→        1: s[1],
   680→   680→        2: { status: 'pending' },
   681→   681→        3: { status: 'pending' },
   682→   682→        4: { status: 'pending' },
   683→   683→      }))
   684→   684→      toast({
   685→   685→        title: 'Command dispatched',
   686→   686→        description: `PublishGame appended to stream GameAggregate:${gameId}`,
   687→   687→      })
   688→   688→    } catch (e) {
   689→   689→      const msg = e instanceof Error ? e.message : 'Unknown error'
   690→   690→      updateStep(1, { status: 'error', error: msg })
   691→   691→      toast({ title: 'Command failed', description: msg, variant: 'destructive' })
   692→   692→    }
   693→   693→  }
   694→   694→
   695→   695→  // Step 2: Process outbox
   696→   696→  const handleOutbox = async () => {
   697→   697→    updateStep(2, { status: 'running', error: undefined, result: undefined })
   698→   698→    try {
   699→   699→      const res = await fetch('/api/workers/outbox', { method: 'POST' })
   700→   700→      const data = (await res.json()) as OutboxResponse
   701→   701→      updateStep(2, {
   702→   702→        status: 'success',
   703→   703→        result: data,
   704→   704→        timestamp: new Date().toISOString(),
   705→   705→      })
   706→   706→      toast({
   707→   707→        title: 'Outbox processed',
   708→   708→        description: `Published ${data.published} event(s).`,
   709→   709→      })
   710→   710→    } catch (e) {
   711→   711→      const msg = e instanceof Error ? e.message : 'Unknown error'
   712→   712→      updateStep(2, { status: 'error', error: msg })
   713→   713→      toast({ title: 'Outbox failed', description: msg, variant: 'destructive' })
   714→   714→    }
   715→   715→  }
   716→   716→
   717→   717→  // Step 3: Run projections
   718→   718→  const handleProjections = async () => {

   720→   720→    try {
   721→   721→      const res = await fetch('/api/workers/projections', { method: 'POST' })
   722→   722→      const data = (await res.json()) as ProjectionsResponse
   723→   723→      updateStep(3, {
   724→   724→        status: 'success',
   725→   725→        result: data,
   726→   726→        timestamp: new Date().toISOString(),
   727→   727→      })
   728→   728→      toast({
   729→   729→        title: 'Projections ran',
   730→   730→        description: `Projected ${data.processed} event(s) into read models.`,
   731→   731→      })
   732→   732→    } catch (e) {
   733→   733→      const msg = e instanceof Error ? e.message : 'Unknown error'

   735→   735→      toast({
   736→   736→        title: 'Projections failed',
   737→   737→        description: msg,
   738→   738→        variant: 'destructive',
   739→   739→      })
   740→   740→    }
   741→   741→  }

   758→   758→      }
   759→   759→      updateStep(4, {

   771→   771→      toast({ title: 'Query failed', description: msg, variant: 'destructive' })
   772→   772→    }
   773→   773→  }

   800→   800→      />

   802→   802→      {/* Flow indicator */}
