   208→// Use global to persist across Next.js dev-mode module reloads.
   209→const globalForContainer = globalThis as unknown as {
   210→  __playliquidContainer?: DIContainer;
   211→  __playliquidContainerBuilding?: Promise<DIContainer>;
   212→};
   213→
   214→let container: DIContainer | null = globalForContainer.__playliquidContainer ?? null;
   215→
   216→/** Build and configure the DI container. */
   217→export async function buildContainer(): Promise<DIContainer> {
   218→  if (container) return container;
   219→
   220→  // Prevent concurrent builds (multiple requests arriving simultaneously)
   221→  if (globalForContainer.__playliquidContainerBuilding) {
   222→    return globalForContainer.__playliquidContainerBuilding;
   223→  }
   224→
   225→  const buildPromise = (async () => {
   226→
   227→  // Register domain events first.
   228→  registerAllEvents();
   229→  registerIdentityEvents();
   230→  registerLaunchEvents();
   231→
   232→  const c = new DIContainer();
   233→
   234→  // ─── Core infrastructure ───────────────────────────────────────────────────
   235→  c.bind(TOKENS.PrismaClient, prisma);
   236→  c.singleton(TOKENS.EventStore, () => new PrismaEventStore());
   237→  c.singleton(TOKENS.SnapshotStore, () => new PrismaSnapshotStore());
   238→  c.singleton(TOKENS.OutboxRepository, () => new OutboxRepository());
   239→  c.singleton(TOKENS.EventBus, () => new InMemoryEventBus());
   240→  c.singleton(TOKENS.Cache, () => new InMemoryCache(getConfig().cache.maxSize));
   241→  c.singleton(TOKENS.MetricsRecorder, () => new InMemoryMetricsRecorder());
   242→  c.singleton(TOKENS.IdempotencyStore, () => new PrismaIdempotencyStore());
   243→  c.singleton(TOKENS.GameReadModelStore, () => new PrismaGameReadModelStore());
   244→  c.singleton(TOKENS.CheckpointStore, () => new CheckpointStore());
   245→  c.singleton(TOKENS.UnitOfWorkFactory, () => new PrismaUnitOfWorkFactory());
   246→
   247→  // ─── Redis platform ────────────────────────────────────────────────────────
   248→  const redisClient = await getRedisClient();
   249→  c.bind(TOKENS.RedisClient, redisClient);
   250→
   251→  // ─── Cache provider (Redis if available, else memory) ─────────────────────
   252→  c.singleton(TOKENS.CacheProvider, (c) => {
   253→    const redis = c.resolve(TOKENS.RedisClient);
   254→    if (redis.backend === 'redis') {
   255→      return new RedisCacheProvider(redis);
   256→    }
   257→    return new MemoryCacheProvider(getConfig().cache.maxSize);
   258→  });
   259→
   260→  // ─── Distributed lock provider ─────────────────────────────────────────────
   261→  c.singleton(TOKENS.LockProvider, (c) => {
   262→    const redis = c.resolve(TOKENS.RedisClient);
   263→    if (redis.backend === 'redis') {
   264→      return new RedisLockProvider(redis);
   265→    }
   266→    return new MemoryLockProvider();
   267→  });
   268→
   269→  // ─── Circuit breaker registry ─────────────────────────────────────────────
   270→  c.singleton(TOKENS.CircuitBreakerRegistry, () => new CircuitBreakerRegistry());
   271→
   272→  // ─── Rate limiter ──────────────────────────────────────────────────────────
   273→  c.singleton(TOKENS.RateLimiter, (c) => {
   274→    const redis = c.resolve(TOKENS.RedisClient);
   275→    if (redis.backend === 'redis') {
   276→      return new RedisRateLimiter(redis);
   277→    }
   278→    return new MemoryRateLimiter();
   279→  });
   280→
   281→  // ─── Message queue + DLQ ───────────────────────────────────────────────────
   282→  c.singleton(TOKENS.MessageQueue, () => new InMemoryQueue());
   283→  c.singleton(TOKENS.DeadLetterQueue, () => new PrismaDeadLetterQueue());
   284→
   285→  // ─── Storage / CDN / Search ────────────────────────────────────────────────
   286→  c.singleton(TOKENS.StorageProvider, () => new LocalStorageProvider(getEnvVar('STORAGE_ROOT') || './storage'));
   287→  c.singleton(TOKENS.CdnProvider, () => new LocalCdnProvider(getEnvVar('CDN_BASE_URL') || ''));
   288→  c.singleton(TOKENS.SearchProvider, () => new InMemorySearchProvider());
   289→
   290→  // ─── Platform services ─────────────────────────────────────────────────────
   291→  c.singleton(TOKENS.FeatureFlagService, () => new CachedFeatureFlagService());
   292→  c.singleton(TOKENS.SecretProvider, () => new EnvironmentSecretProvider());
   293→  c.singleton(TOKENS.ConfigService, () => new DefaultConfigService());
   294→
   295→  // ─── Notifications ─────────────────────────────────────────────────────────
   296→  c.singleton(TOKENS.EmailProvider, () => new ConsoleEmailProvider());
   297→  c.singleton(TOKENS.SmsProvider, () => new ConsoleSmsProvider());
   298→  c.singleton(TOKENS.PushProvider, () => new ConsolePushProvider());
   299→  c.singleton(TOKENS.WebhookEngine, () => new DefaultWebhookEngine());
   300→  c.singleton(TOKENS.SessionStore, () => new InMemorySessionStore());
   301→  c.singleton(TOKENS.JwtService, () => new HmacJwtService(getConfig().auth.secret));
   302→
   303→  // ─── Metrics framework ─────────────────────────────────────────────────────
   304→  c.singleton(TOKENS.MetricsFramework, () => new InMemoryMetricsFramework());
   305→
   306→  // ─── Outbox publisher ──────────────────────────────────────────────────────
   307→  c.singleton(TOKENS.OutboxPublisher, (c) => {
   308→    return new OutboxPublisher(
   309→      c.resolve(TOKENS.OutboxRepository),
   310→      c.resolve(TOKENS.EventBus),
   311→    );
   312→  });
   313→
   314→  // ─── Projection engine ─────────────────────────────────────────────────────
   315→  c.singleton(TOKENS.ProjectionEngine, (c) => {
   316→    const engine = new ProjectionEngine(
   317→      c.resolve(TOKENS.EventStore),
   318→      c.resolve(TOKENS.CheckpointStore),
   319→    );
   320→    engine.register(new GameProjector());
   321→    engine.register(new WalletProjector());
   322→    engine.register(new LeaderboardProjector());
   323→    engine.register(new StatisticsProjector());
   324→    return engine;
   325→  });
   326→
   327→  // ─── Worker registry ───────────────────────────────────────────────────────
   328→  c.singleton(TOKENS.WorkerRegistry, (c) => {
   329→    const registry = new WorkerRegistry();
   330→    const config = getConfig();
   331→
   332→    if (config.featureFlags.outboxWorker) {
   333→      registry.register(new OutboxWorker(
   334→        c.resolve(TOKENS.OutboxPublisher),
   335→        config.outbox.pollIntervalMs,
   336→      ));
   337→    }
   338→
   339→    if (config.featureFlags.projectionWorker) {
   340→      registry.register(new ProjectionWorker(
   341→        c.resolve(TOKENS.ProjectionEngine),
   342→        config.projections.pollIntervalMs,
   343→      ));
   344→    }
   345→
   346→    registry.register(new CleanupWorker(60000)); // every 60s
   347→    registry.register(new AnalyticsWorker(c.resolve(TOKENS.EventStore), 30000)); // every 30s
   348→
   349→    return registry;
   350→  });
   351→
   352→  // ─── Scheduler ─────────────────────────────────────────────────────────────
   353→  c.singleton(TOKENS.Scheduler, (c) => {
   354→    return new InMemoryScheduler(c.resolve(TOKENS.LockProvider));
   355→  });
   356→
   357→  // ─── M3: Identity Infrastructure ───────────────────────────────────────────
   358→  c.singleton('PasswordHasher', () => new ScryptPasswordHasher());
   359→  c.singleton('UserRepository', (c) => new UserRepositoryImpl(
   360→    c.resolve(TOKENS.EventStore), c.resolve(TOKENS.SnapshotStore), c.resolve(TOKENS.OutboxRepository),
   361→  ));
   362→  c.singleton('OrganizationRepository', (c) => new OrganizationRepositoryImpl(
   363→    c.resolve(TOKENS.EventStore), c.resolve(TOKENS.SnapshotStore), c.resolve(TOKENS.OutboxRepository),
   364→  ));
   365→  c.singleton('RoleRepository', () => new PrismaRoleRepository());
   366→  c.singleton('PermissionRepository', () => new PrismaPermissionRepository());
   367→  c.singleton('ApiKeyRepository', () => new PrismaApiKeyRepository());
   368→  c.singleton('AuditLogRepository', () => new PrismaAuditLogRepository());
   369→  c.singleton('DeviceRepository', () => new PrismaDeviceRepository());
   370→  c.singleton('WaitlistRepository', () => new PrismaWaitlistRepository());
   371→
   372→  // M3: Authorization engine
   373→  c.singleton('RbacEngine', () => new RbacEngine());
   374→  c.singleton('AbacEngine', () => new AbacEngine());
   375→  c.singleton('PolicyEngine', (c) => new PolicyEngine(
   376→    c.resolve('RbacEngine'), c.resolve('AbacEngine'),
   377→  ));
   378→  c.singleton('RiskEngine', () => new RiskEngine());
   379→  c.singleton('TokenStore', () => new InMemoryTokenStore());
   380→  c.singleton('EmailService', () => new ConsoleEmailService());
   381→  c.singleton('UserReadModelStore', () => new PrismaUserReadModelStore());
   382→  c.singleton('OrganizationReadModelStore', () => new PrismaOrganizationReadModelStore());
   383→
   384→  // M3: Register identity projectors on the projection engine
   385→  c.singleton(TOKENS.ProjectionEngine, (c) => {
   386→    const engine = new ProjectionEngine(
   387→      c.resolve(TOKENS.EventStore),
   388→      c.resolve(TOKENS.CheckpointStore),
   389→    );
   390→    // M1 projectors
   391→    engine.register(new GameProjector());
   392→    engine.register(new WalletProjector());
   393→    engine.register(new LeaderboardProjector());
   394→    engine.register(new StatisticsProjector());
   395→    // M3 identity projectors
   396→    engine.register(new UserProfileProjector());
   397→    engine.register(new OrganizationProjector());
   398→    engine.register(new AuditLogProjector());
   399→    engine.register(new ApiKeyProjector());
   400→    // Launch projectors
   401→    engine.register(new BetaCohortProjector());
   402→    return engine;
   403→  });
   404→
   405→  // ─── Launch & Scale Infrastructure ─────────────────────────────────────────
   406→  c.singleton('BetaCohortRepository', (c) => new BetaCohortRepositoryImpl(
   407→    c.resolve(TOKENS.EventStore), c.resolve(TOKENS.SnapshotStore), c.resolve(TOKENS.OutboxRepository),
   408→  ));
   409→  c.singleton('FeedbackRepository', () => new PrismaFeedbackRepository());
   410→  c.singleton('ValidationRunRepository', () => new PrismaValidationRunRepository());
   411→  c.singleton('ReconciliationRepository', () => new PrismaReconciliationRepository());
   412→  c.singleton('SessionReplayRepository', () => new PrismaSessionReplayRepository());
   413→  c.singleton('BugRepository', () => new PrismaBugRepository());
   414→  c.singleton('PerformanceMetricRepository', () => new PrismaPerformanceMetricRepository());
   415→  c.singleton('ReconciliationSource', (c) => new PrismaReconciliationSource(c.resolve(TOKENS.EventStore)));
   416→  c.singleton('BetaCohortReadModelStore', () => new PrismaBetaCohortReadModelStore());
   417→  c.singleton('ReconciliationService', (c) => new ReconciliationService(c.resolve('ReconciliationSource')));
   418→  c.singleton('ValidationSuiteRunner', (c) => {
   419→    const runner = new ValidationSuiteRunner();
   420→    const suites = createPlatformValidationSuites({
   421→      eventStore: c.resolve(TOKENS.EventStore),
   422→      reconciliationService: c.resolve('ReconciliationService'),
   423→      rateLimiter: c.resolve(TOKENS.RateLimiter),
   424→      storageProvider: c.resolve(TOKENS.StorageProvider),
   425→      sessionReplayRepo: c.resolve('SessionReplayRepository'),
   426→    });
   427→    for (const suite of suites) runner.registerSuite(suite);
   428→    return runner;
   429→  });
   430→
   431→  // ─── Backup ────────────────────────────────────────────────────────────────
   432→  c.singleton(TOKENS.BackupProvider, () => new LocalBackupProvider());
   433→
   434→  // ─── Disaster recovery ─────────────────────────────────────────────────────
   435→  c.singleton(TOKENS.DisasterRecoveryService, (c) => {
   436→    return new DefaultDisasterRecoveryService(
   437→      c.resolve(TOKENS.ProjectionEngine),
