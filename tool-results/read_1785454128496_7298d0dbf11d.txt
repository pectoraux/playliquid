   301→async function apiGet<T>(path: string): Promise<T> {
   302→  const res = await fetch(path, { cache: 'no-store' })
   303→  const json = (await res.json().catch(() => ({}))) as { ok?: boolean; data?: T; error?: string }
   304→  if (!res.ok || !json.ok) {
   305→    throw new Error(json.error || `Request failed (${res.status})`)
   306→  }
   307→  return json.data as T
   308→}
   309→
   310→async function apiPost<T>(path: string, payload: Record<string, unknown>): Promise<T> {
   311→  const res = await fetch(path, {
   312→    method: 'POST',
   313→    headers: { 'Content-Type': 'application/json' },
   314→    body: JSON.stringify({ payload }),
   315→  })
   316→  const json = (await res.json().catch(() => ({}))) as { ok?: boolean; data?: T; error?: string; code?: string }
   317→  if (!res.ok || !json.ok) {
   318→    throw new Error(json.error || `Request failed (${res.status})`)
   319→  }
   320→  return json.data as T
   321→}

   325→function formatRelativeTime(input: string | null | undefined): string {
   326→  if (!input) return '—'
   327→  const t = new Date(input).getTime()
   328→  if (Number.isNaN(t)) return String(input)
   329→  const diff = Date.now() - t
   330→  const sec = Math.round(diff / 1000)
   331→  if (sec < 5) return 'just now'
   332→  if (sec < 60) return `${sec} seconds ago`
   333→  const min = Math.round(sec / 60)
   334→  if (min < 60) return `${min} minute${min === 1 ? '' : 's'} ago`
   335→  const hr = Math.round(min / 60)
   336→  if (hr < 24) return `${hr} hour${hr === 1 ? '' : 's'} ago`
   337→  const day = Math.round(hr / 24)
   338→  if (day < 30) return `${day} day${day === 1 ? '' : 's'} ago`
   339→  const month = Math.round(day / 30)
   340→  if (month < 12) return `${month} month${month === 1 ? '' : 's'} ago`
   341→  const year = Math.round(month / 12)
   342→  return `${year} year${year === 1 ? '' : 's'} ago`
   343→}

   345→function statusTone(status: string): { className: string; label: string } {
   346→  const s = status.toLowerCase()
   347→  if (['active', 'approved', 'converted', 'email_verified'].includes(s)) {
   348→    return {
   349→      className: 'border-emerald-500/40 bg-emerald-500/10 text-emerald-300',
   350→      label: status,
   351→    }
   352→  }
   353→  if (['pending', 'waitlist'].includes(s)) {
   354→    return {
   355→      className: 'border-amber-500/40 bg-amber-500/10 text-amber-300',
   356→      label: status,
   357→    }
   358→  }
   359→  if (['rejected', 'suspended', 'deleted', 'revoked'].includes(s)) {
   360→    return {
   361→      className: 'border-rose-500/40 bg-rose-500/10 text-rose-300',
   362→      label: status,
   363→    }
   364→  }
   365→  return {
   366→    className: 'border-slate-600 bg-slate-800/40 text-slate-300',
   367→    label: status,
   368→  }
   369→}
